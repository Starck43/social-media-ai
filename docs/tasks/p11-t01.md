```python
import pathlib,re,subprocess,urllib.parse
base='00f530702f924d6a677d5c565a389b1e44cab88f'
paths=subprocess.check_output(['git','diff','--name-only',base],text=True).splitlines()
if pathlib.Path('REFACTORING_PLAN.md').exists() and 'REFACTORING_PLAN.md' not in paths: paths.append('REFACTORING_PLAN.md')
fail=[]; links=0
for name in paths:
 p=pathlib.Path(name)
 if p.suffix!='.md':fail.append((name,'non-doc changed'));continue
 text=re.sub(r'```.*?```','',p.read_text(),flags=re.S)
 for target in re.findall(r'\[[^\]\n]+\]\(([^)\s]+)\)',text):
  u=urllib.parse.urlsplit(target)
  if u.scheme or u.netloc:continue
  q=(p.parent/urllib.parse.unquote(u.path)).resolve() if u.path else p.resolve()
  links+=1
  if not q.exists():fail.append((name,target,'missing path'));continue
  if u.fragment and q.suffix=='.md':
   headings=re.findall(r'^#{1,6}\s+(.+)$',q.read_text(),flags=re.M)
   slugs={re.sub(r'[^\w\- ]','',re.sub(r'\[([^]]+)\]\([^)]+\)',r'\1',h).lower()).replace(' ','-') for h in headings}
   if urllib.parse.unquote(u.fragment) not in slugs and not re.search(r'(?:id|name)=["\']'+re.escape(u.fragment),q.read_text()):fail.append((name,target,'missing heading'))
print('changed Markdown:',len(paths),'local link/anchor targets:',links,'failures:',len(fail))
for f in fail:print(f)
raise SystemExit(bool(fail))
```
